     3→/**
     4→ * WebMCP tool definitions for the Meridian Ledger newsroom.
     5→ *
     6→ * These tools are the machine-facing surface of the exact same REST API the
     7→ * human UI uses, so an external browser agent mutating a story through
     8→ * `update_story_status` changes the same state a human editor sees — and
     9→ * vice versa. After every mutating call we emit a `newsroom:data-changed`
    10→ * window event so the on-page UI refreshes immediately.
    11→ */
    12→

    14→import type { ModelContextTool } from "@/types/webmcp";

    17→const DATA_CHANGED_EVENT = "newsroom:data-changed";
    18→
    19→function notifyDataChanged() {
    20→  window.dispatchEvent(new CustomEvent(DATA_CHANGED_EVENT));
    21→}

    23→/** Compact story projection used in list responses to keep outputs small. */
    24→function storySummary(s: Awaited<ReturnType<typeof api.fetchStories>>["stories"][number]) {
    25→  return {
    26→    id: s.id,
    27→    slug: s.slug,
    28→    title: s.title,
    29→    status: s.status,
    30→    desk: s.desk,
    31→    author: s.author,
    32→    assignee: s.assignee,
    33→    priority: s.priority,
    34→    wordCount: s.wordCount,
    35→    mediaCount: s.media.length,
    36→    hasExpiredMedia: s.media.some((m) => m.media.rights.status === "EXPIRED"),
    37→    handoffCount: s.handoffCount,
    38→    updatedAt: s.updatedAt,
    39→  };
    40→}
    41→
    42→function mediaSummary(m: Awaited<ReturnType<typeof api.fetchMedia>>["media"][number]) {
    43→  return {
    44→    id: m.id,
    45→    fileName: m.fileName,
    46→    url: m.url,
    47→    title: m.title,
    48→    photographer: m.photographer,
    49→    source: m.source,
    50→    licenseType: m.licenseType,
    51→    rightsStatus: m.rights.status,
    52→    expiresAt: m.rights.expiresAt,
    53→    allowedChannels: m.allowedChannels,
    54→    webAllowed: m.rights.webAllowed,
    55→    creditLine: m.creditLine,
    56→    attachedToCount: m.attachedToCount,
    57→  };
    58→}
    59→
    60→const ok = (data: Record<string, unknown>) => ({ ok: true as const, ...data });
    61→const fail = (error: string, extra: Record<string, unknown> = {}) => ({
    62→  ok: false as const,
    63→  error,
    64→  ...extra,
    65→});
    66→
    67→function asString(v: unknown): string | undefined {
    68→  return typeof v === "string" && v.length > 0 ? v : undefined;
    69→}
    70→
    71→function requireString(input: Record<string, unknown>, key: string): string {
    72→  const v = input[key];
    73→  if (typeof v !== "string" || v.trim() === "") {
    74→    throw new Error(`Missing required string parameter "${key}".`);
    75→  }
    76→  return v.trim();
    77→}

    79→function run<T>(fn: () => Promise<T>): Promise<T> {
    80→  return fn();
    81→}
