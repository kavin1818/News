    60→    60→const ok = (data: Record<string, unknown>) => ({ ok: true as const, ...data });
    61→    61→const fail = (error: string, extra: Record<string, unknown> = {}) => ({

    67→    67→function asString(v: unknown): string | undefined {
    68→    68→  return typeof v === "string" && v.length > 0 ? v : undefined;
    69→    69→}
    70→    70→
    71→    71→function requireString(input: Record<string, unknown>, key: string): string {
    72→    72→  const v = input[key];
    73→    73→  if (typeof v !== "string" || v.trim() === "") {
    74→    74→    throw new Error(`Missing required string parameter "${key}".`);
    75→    75→  }
    76→    76→  return v.trim();
    77→    77→}
