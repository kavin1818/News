    62→    62→const ok = (data: Record<string, unknown>) => ({ ok: true as const, ...data });
    63→    63→const fail = (error: string, extra: Record<string, unknown> = {}) => ({

    68→    68→
    69→    69→function asString(v: unknown): string | undefined {
    70→    70→  return typeof v === "string" && v.length > 0 ? v : undefined;
    71→    71→}
    72→    72→
    73→    73→function requireString(input: Record<string, unknown>, key: string): string {
    74→    74→  const v = input[key];
    75→    75→  if (typeof v !== "string" || v.trim() === "") {
    76→    76→    throw new Error(`Missing required string parameter "${key}".`);
    77→    77→  }
    78→    78→  return v.trim();
    79→    79→}
