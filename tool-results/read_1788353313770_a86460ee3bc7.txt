    67→export function StoryDesk({ onSendToHandoff }: { onSendToHandoff: (id: string) => void }) {
    68→  const stories = useNewsroom((s) => s.stories);
    69→  const loading = useNewsroom((s) => s.storiesLoading);
    70→  const [selectedId, setSelectedId] = useState<string | null>(null);
    71→  const [query, setQuery] = useState("");
    72→  const [statusFilter, setStatusFilter] = useState("any");
    73→  const [deskFilter, setDeskFilter] = useState("any");
    74→  const [newOpen, setNewOpen] = useState(false);
    75→
    76→  const filtered = useMemo(() => {
    77→    const q = query.trim().toLowerCase();
    78→    return stories.filter((s) => {
    79→      if (statusFilter !== "any" && s.status !== statusFilter) return false;
    80→      if (deskFilter !== "any" && s.desk !== deskFilter) return false;
    81→      if (q) {
    82→        const hay = `${s.title} ${s.summary} ${s.author} ${s.assignee ?? ""} ${s.slug}`.toLowerCase();
    83→        if (!hay.includes(q)) return false;
    84→      }
    85→      return true;
    86→    });
    87→  }, [stories, query, statusFilter, deskFilter]);
    88→
    89→  const selected =
    90→    stories.find((s) => s.id === selectedId) ?? filtered[0] ?? null;
    91→
    92→  const statusCounts = useMemo(() => {
    93→    const counts: Record<string, number> = { any: stories.length };
    94→    for (const s of stories) counts[s.status] = (counts[s.status] ?? 0) + 1;
    95→    return counts;
    96→  }, [stories]);
    97→
    98→  return (
    99→    <div className="rise-in">
   100→      <div className="mb-5 flex flex-wrap items-end justify-between gap-3">
   101→        <div>
   102→          <h2 className="font-headline text-2xl font-bold">Story Desk</h2>
   103→          <p className="text-sm text-muted-foreground">
   104→            Write, assign, review and move stories through the editorial workflow.
   105→          </p>
   106→        </div>
   107→        <Button onClick={() => setNewOpen(true)} className="gap-1.5">
   108→          <Plus className="h-4 w-4" /> New Story
   109→        </Button>
   110→      </div>
   111→
   112→      <div className="grid gap-6 lg:grid-cols-5">
   113→        {/* ------------------------------------------------------ list pane */}
   114→        <section aria-label="Story list" className="lg:col-span-2">
   115→          <div className="flex gap-2">
   116→            <div className="relative flex-1">
   117→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   118→              <Input
   119→                value={query}
   120→                onChange={(e) => setQuery(e.target.value)}
   121→                placeholder="Search stories…"
   122→                className="pl-8"
   123→                aria-label="Search stories"
   124→              />
   125→            </div>
   126→            <Select value={deskFilter} onValueChange={setDeskFilter}>
   127→              <SelectTrigger className="w-[140px]" aria-label="Filter by desk">
   128→                <SelectValue placeholder="All desks" />
   129→              </SelectTrigger>
   130→              <SelectContent>
   131→                <SelectItem value="any">All desks</SelectItem>
   132→                {DESKS.map((d) => (
   133→                  <SelectItem key={d} value={d}>
   134→                    {d}
   135→                  </SelectItem>
   136→                ))}
   137→              </SelectContent>
   138→            </Select>
   139→          </div>
   140→
   141→          <div className="scrollbar-slim mt-3 flex flex-wrap gap-1.5 pb-1">
   142→            {[
   143→              { id: "any", label: "All" },
   144→              { id: "DRAFT", label: "Draft" },
   145→              { id: "IN_REVIEW", label: "In Review" },
   146→              { id: "REVISION_REQUESTED", label: "Revision" },
   147→              { id: "APPROVED", label: "Approved" },
   148→              { id: "PUBLISHED", label: "Published" },
   149→            ].map((st) => (
   150→              <button
   151→                key={st.id}
   152→                onClick={() => setStatusFilter(st.id)}
   153→                className={cn(
   154→                  "whitespace-nowrap rounded-full border px-2.5 py-1 text-xs font-medium transition-all active:scale-95",
   155→                  statusFilter === st.id
   156→                    ? "border-primary bg-primary text-primary-foreground shadow-sm"
   157→                    : "bg-card text-muted-foreground hover:border-foreground/30 hover:text-foreground"
   158→                )}
   159→              >
   160→                {st.label}
   161→                <span className={cn("ml-1", statusFilter === st.id ? "text-primary-foreground/80" : "text-muted-foreground/70")}>
   162→                  {statusCounts[st.id] ?? 0}
   163→                </span>
   164→              </button>
   165→            ))}
   166→          </div>
   167→
   168→          <div className="scrollbar-slim mt-3 max-h-[calc(100vh-16rem)] space-y-2 overflow-y-auto pr-1 lg:max-h-[calc(100vh-19rem)]">
   169→            {loading && stories.length === 0 && (
   170→              <div className="space-y-2">
   171→                {[0, 1, 2, 3].map((i) => (
   172→                  <div key={i} className="flex items-start gap-3 rounded-lg border p-3">
   173→                    <Skeleton className="h-14 w-20 rounded-md" />
   174→                    <div className="flex-1 space-y-2">
   175→                      <Skeleton className="h-3.5 w-4/5" />
   176→                      <Skeleton className="h-3 w-1/3" />
   177→                      <Skeleton className="h-2.5 w-1/2" />
   178→                    </div>
   179→                  </div>
   180→                ))}
   181→              </div>
   182→            )}
   183→            {!loading && filtered.length === 0 && (
   184→              <div className="flex flex-col items-center gap-3 rounded-lg border border-dashed bg-card/50 py-10 text-center">
   185→                <FileText className="h-7 w-7 text-muted-foreground/40" />
   186→                <p className="text-sm text-muted-foreground">
   187→                  No stories match the current search and filters.
   188→                </p>
   189→                {(query || statusFilter !== "any" || deskFilter !== "any") && (
   190→                  <Button
   191→                    size="sm"
   192→                    variant="outline"
   193→                    onClick={() => {
   194→                      setQuery("");
   195→                      setStatusFilter("any");
   196→                      setDeskFilter("any");
   197→                    }}
   198→                  >
   199→                    <X className="mr-1.5 h-3.5 w-3.5" /> Clear filters
   200→                  </Button>
   201→                )}
   202→              </div>
   203→            )}
   204→            {filtered.map((story) => (
   205→              <StoryListItem
   206→                key={story.id}
   207→                story={story}
   208→                active={story.id === selectedId}
   209→                onSelect={() => setSelectedId(story.id)}
   210→              />
   211→            ))}
   212→          </div>
   213→        </section>
   214→
   215→        {/* ----------------------------------------------------- detail pane */}
   216→        <section aria-label="Story detail" className="lg:col-span-3">
   217→          {selected ? (
   218→            <StoryDetail
   219→              key={selected.id}
   220→              story={selected}
   221→              onSendToHandoff={onSendToHandoff}
   222→              onDuplicated={setSelectedId}
   223→            />
   224→          ) : (
   225→            <div className="flex h-64 flex-col items-center justify-center gap-2 rounded-lg border border-dashed bg-card/50 text-center">
   226→              <FileText className="h-8 w-8 text-muted-foreground/50" />
   227→              <p className="text-sm text-muted-foreground">
   228→                Select a story from the desk, or create a new one.
   229→              </p>
   230→            </div>
   231→          )}
   232→        </section>
   233→      </div>
   234→
   235→      <NewStoryDialog open={newOpen} onOpenChange={setNewOpen} onCreated={(id) => setSelectedId(id)} />
   236→    </div>
   237→  );
   238→}

   244→function StoryListItem({
   245→  story,
   246→  active,
   247→  onSelect,
   248→}: {
   249→  story: StoryDTO;
   250→  active: boolean;
   251→  onSelect: () => void;
   252→}) {
   253→  const primary = story.media.find((m) => m.isPrimary) ?? story.media[0];
   254→  const hasExpired = story.media.some((m) => m.media.rights.status === "EXPIRED");
   255→
   256→  return (
   257→    <button
   258→      onClick={onSelect}
   259→      aria-current={active ? "true" : undefined}
   260→      className={cn(
   261→        "group flex w-full items-start gap-3 rounded-lg border p-3 text-left transition-all active:scale-[0.99]",
   262→        active
   263→          ? "border-primary/60 bg-card shadow-[0_1px_0_0_var(--primary)] ring-1 ring-primary/40"
   264→          : "border-border bg-card hover:border-foreground/25 hover:shadow-sm"
   265→      )}
   266→    >
   267→      <div className="relative h-14 w-20 shrink-0 overflow-hidden rounded-md bg-muted">
   268→        {primary ? (
   269→          <Image
   270→            src={primary.media.url}
   271→            alt={primary.media.title}
   272→            fill
   273→            sizes="80px"
   274→            className="object-cover"
   275→          />
   276→        ) : (
   277→          <div className="flex h-full w-full items-center justify-center">
   278→            <ImageIcon className="h-4 w-4 text-muted-foreground/40" />
   279→          </div>
   280→        )}
   281→      </div>
   282→      <div className="min-w-0 flex-1">
   283→        <p className="line-clamp-2 font-headline text-[15px] font-semibold leading-snug transition-colors group-hover:text-primary">
   284→          {story.title}
   285→        </p>
   286→        <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   287→          <StatusBadge status={story.status} />
   288→          {(story.priority === "HIGH" || story.priority === "URGENT") && (
   289→            <Badge
   290→              variant="outline"
   291→              className={cn(
   292→                "px-1.5 py-0 text-[10px] font-bold",
   293→                story.priority === "URGENT"
   294→                  ? "border-red-300 bg-red-50 text-red-700"
   295→                  : "border-amber-300 bg-amber-50 text-amber-700"
   296→              )}
   297→            >
   298→              {story.priority}
   299→            </Badge>
   300→          )}
   301→        </div>
   302→        <p className="mt-1.5 flex flex-wrap items-center gap-x-2 text-xs text-muted-foreground">
   303→          <span className="font-medium text-foreground/70">{story.desk}</span>
   304→          <span aria-hidden>·</span>
   305→          <span>{story.author}</span>
   306→          <span aria-hidden>·</span>
   307→          <span>{fmtRelative(story.updatedAt)}</span>
   308→          {hasExpired && (
   309→            <span className="inline-flex items-center gap-0.5 font-medium text-red-700">
   310→              <AlertTriangle className="h-3 w-3" /> expired media
   311→            </span>
   312→          )}
   313→          {story.scheduledFor &&
   314→            story.status !== "PUBLISHED" &&
   315→            deadlineState(story.scheduledFor) === "past" && (
   316→              <span className="inline-flex items-center gap-0.5 font-medium text-red-700">
   317→                <CalendarClock className="h-3 w-3" /> past due
   318→              </span>
   319→            )}
   320→          {story.scheduledFor &&
   321→            story.status !== "PUBLISHED" &&
   322→            deadlineState(story.scheduledFor) === "soon" && (
   323→              <span className="inline-flex items-center gap-0.5 font-medium text-amber-700">
   324→                <CalendarClock className="h-3 w-3" /> due soon
   325→              </span>
   326→            )}
   327→        </p>
   328→      </div>
   329→    </button>
   330→  );
   331→}

   337→function StoryDetail({
   338→  story,
   339→  onSendToHandoff,
   340→  onDuplicated,
   341→}: {
   342→  story: StoryDTO;
   343→  onSendToHandoff: (id: string) => void;
   344→  onDuplicated: (id: string) => void;
   345→}) {
   346→  const { toast } = useToast();
   347→  const setStoryStatus = useNewsroom((s) => s.setStoryStatus);
   348→  const approveStory = useNewsroom((s) => s.approveStory);
   349→  const duplicateStory = useNewsroom((s) => s.duplicateStory);
   350→  const [busyAction, setBusyAction] = useState<string | null>(null);
   351→  const [editing, setEditing] = useState(false);
   352→  const [attachOpen, setAttachOpen] = useState(false);
   353→  const [previewOpen, setPreviewOpen] = useState(false);
   354→  const [replaceMediaId, setReplaceMediaId] = useState<string | null>(null);
   355→
   356→  const deadline = deadlineState(story.scheduledFor);
   357→  const deadlineOverdue =
   358→    deadline === "past" && story.status !== "PUBLISHED";
   359→
   360→  const runAction = async (id: string, fn: () => Promise<unknown>, success: string) => {
   361→    setBusyAction(id);
   362→    try {
   363→      await fn();
   364→      toast({ title: success });
   365→    } catch (e) {
   366→      toast({
   367→        title: "Action failed",
   368→        description: e instanceof Error ? e.message : "Unknown error",
   369→        variant: "destructive",
   370→      });
   371→    } finally {
   372→      setBusyAction(null);
   373→    }
   374→  };
   375→
   376→  return (
   377→    <article className="rise-in rounded-lg border bg-card">
   378→      {/* header */}
   379→      <div className="border-b p-5">
   380→        <div className="flex flex-wrap items-center gap-2">
   381→          <StatusBadge status={story.status} />
   382→          <Badge variant="outline" className="text-xs">
   383→            {story.desk} desk
   384→          </Badge>
   385→          {story.priority !== "NORMAL" && (
   386→            <Badge variant="outline" className="text-xs">
   387→              {story.priority}
   388→            </Badge>
   389→          )}
   390→          {story.handoffCount > 0 && (
   391→            <Badge variant="outline" className="gap-1 border-primary/40 text-[11px] text-primary">
   392→              <Send className="h-3 w-3" /> {story.handoffCount} handoff(s)
   393→            </Badge>
   394→          )}
   395→          {story.scheduledFor && (
   396→            <Badge
   397→              variant="outline"
   398→              className={cn(
   399→                "gap-1 text-[11px]",
   400→                deadlineOverdue
   401→                  ? "border-red-300 bg-red-50 font-medium text-red-700"
   402→                  : deadline === "soon" && story.status !== "PUBLISHED"
   403→                    ? "border-amber-300 bg-amber-50 font-medium text-amber-800"
   404→                    : "text-muted-foreground"
   405→              )}
   406→            >
   407→              <CalendarClock className="h-3 w-3" />
   408→              {story.status !== "PUBLISHED" && deadline === "past"
   409→                ? "past due "
   410→                : story.status !== "PUBLISHED" && deadline === "soon"
   411→                  ? "due soon "
   412→                  : "target "}
   413→              {fmtDate(story.scheduledFor)}
   414→            </Badge>
   415→          )}
   416→        </div>
   417→
   418→        <h3 className="mt-3 font-headline text-2xl font-bold leading-tight">
   419→          {story.title}
   420→        </h3>
   421→        {story.summary && (
   422→          <p className="mt-2 border-l-2 border-primary/50 pl-3 text-sm italic leading-relaxed text-muted-foreground">
   423→            {story.summary}
   424→          </p>
   425→        )}
   426→
   427→        <dl className="mt-4 grid grid-cols-2 gap-x-4 gap-y-1.5 text-xs sm:grid-cols-4">
   428→          <Meta label="Byline" value={story.author} />
   429→          <Meta label="Desk editor" value={story.assignee ?? "Unassigned"} warn={!story.assignee} />
   430→          <Meta label="Slug" value={story.slug} mono />
   431→          <Meta label="Words" value={String(story.wordCount)} />
   432→          <Meta label="Created" value={fmtDateTime(story.createdAt)} />
   433→          <Meta label="Updated" value={fmtRelative(story.updatedAt)} />
   434→          {story.publishedAt && <Meta label="Published" value={fmtDateTime(story.publishedAt)} />}
   435→        </dl>
   436→      </div>
   437→
   438→      {/* workflow actions */}
   439→      <div className="flex flex-wrap items-center gap-2 border-b bg-muted/40 px-5 py-3">
   440→        {story.status === "DRAFT" && (
   441→          <Button
   442→            size="sm"
   443→            disabled={busyAction !== null}
   444→            onClick={() =>
   445→              runAction(
   446→                "submit",
   447→                () => setStoryStatus(story.id, "IN_REVIEW"),
   448→                "Story submitted for review"
   449→              )
   450→            }
   451→          >
   452→            <Eye className="mr-1.5 h-3.5 w-3.5" /> Submit for Review
   453→          </Button>
   454→        )}
   455→        {story.status === "IN_REVIEW" && (
   456→          <>
   457→            <Button
   458→              size="sm"
   459→              className="bg-emerald-700 hover:bg-emerald-800"
   460→              disabled={busyAction !== null}
   461→              onClick={() => runAction("approve", () => approveStory(story.id), "Story approved")}
   462→            >
   463→              <CheckCheck className="mr-1.5 h-3.5 w-3.5" /> Approve
   464→            </Button>
   465→            <Button
   466→              size="sm"
   467→              variant="outline"
   468→              disabled={busyAction !== null}
   469→              onClick={() =>
   470→                runAction(
   471→                  "revise",
   472→                  () => setStoryStatus(story.id, "REVISION_REQUESTED"),
   473→                  "Revision requested"
   474→                )
   475→              }
   476→            >
   477→              <Undo2 className="mr-1.5 h-3.5 w-3.5" /> Request Revision
   478→            </Button>
   479→            <Button
   480→              size="sm"
   481→              variant="ghost"
   482→              disabled={busyAction !== null}
   483→              onClick={() => runAction("draft", () => setStoryStatus(story.id, "DRAFT"), "Sent back to Draft")}
   484→            >
   485→              <PenLine className="mr-1.5 h-3.5 w-3.5" /> Back to Draft
   486→            </Button>
   487→          </>
   488→        )}
   489→        {story.status === "REVISION_REQUESTED" && (
   490→          <>
   491→            <Button
   492→              size="sm"
   493→              disabled={busyAction !== null}
   494→              onClick={() =>
   495→                runAction("resubmit", () => setStoryStatus(story.id, "IN_REVIEW"), "Resubmitted for review")
   496→              }
   497→            >
   498→              <Eye className="mr-1.5 h-3.5 w-3.5" /> Resubmit for Review
   499→            </Button>
   500→            <Button
   501→              size="sm"
   502→              variant="ghost"
   503→              disabled={busyAction !== null}
   504→              onClick={() => runAction("draft", () => setStoryStatus(story.id, "DRAFT"), "Sent back to Draft")}
   505→            >
   506→              <PenLine className="mr-1.5 h-3.5 w-3.5" /> Back to Draft
   507→            </Button>
   508→          </>
   509→        )}
   510→        {story.status === "APPROVED" && (
   511→          <>
   512→            <Button size="sm" className="gap-1.5" disabled={busyAction !== null} onClick={() => onSendToHandoff(story.id)}>
   513→              <Send className="h-3.5 w-3.5" /> Take to Handoff
   514→            </Button>
   515→            <Button
   516→              size="sm"
   517→              variant="outline"
   518→              disabled={busyAction !== null}
   519→              onClick={() => runAction("reopen", () => setStoryStatus(story.id, "IN_REVIEW"), "Review reopened")}
   520→            >
   521→              <Undo2 className="mr-1.5 h-3.5 w-3.5" /> Reopen Review
   522→            </Button>
   523→          </>
   524→        )}
   525→        {story.status === "PUBLISHED" && (
   526→          <p className="flex items-center gap-2 text-sm text-muted-foreground">
   527→            <CalendarClock className="h-4 w-4 text-primary" />
   528→            Published to the web — handed off {fmtRelative(story.publishedAt ?? story.updatedAt)}.
   529→          </p>
   530→        )}
   531→
   532→        <div className="ml-auto flex items-center gap-1">
   533→          <Button
   534→            size="sm"
   535→            variant="ghost"
   536→            className="gap-1.5"
   537→            disabled={busyAction !== null}
   538→            title="Pitch a follow-up story from this one"
   539→            onClick={async () => {
   540→              setBusyAction("duplicate");
   541→              try {
   542→                const created = await duplicateStory(story.id);
   543→                toast({
   544→                  title: "Follow-up pitched",
   545→                  description: `“${created.title}” created as a Draft on the ${created.desk} desk.`,
   546→                });
   547→                onDuplicated(created.id);
   548→              } catch (e) {
   549→                toast({
   550→                  title: "Duplicate failed",
   551→                  description: e instanceof Error ? e.message : "Unknown error",
   552→                  variant: "destructive",
   553→                });
   554→              } finally {
   555→                setBusyAction(null);
   556→              }
   557→            }}
   558→          >
   559→            <CopyPlus className="h-3.5 w-3.5" /> Follow-up
   560→          </Button>
   561→          <Button

   603→        <Separator className="my-5" />

   605→        {/* media */}
   606→        <StoryMediaSection

   629→    </article>

   829→function AttachMediaDialog({
   830→  story,
   831→  open,
   832→  onOpenChange,
   833→}: {
   834→  story: StoryDTO;
   835→  open: boolean;
   836→  onOpenChange: (v: boolean) => void;
   837→}) {
   838→  const media = useNewsroom((s) => s.media);
   839→  const attachMedia = useNewsroom((s) => s.attachMedia);
   840→  const { toast } = useToast();
   841→  const [query, setQuery] = useState("");
   842→  const [caption, setCaption] = useState("");
   843→  const [selectedId, setSelectedId] = useState<string | null>(null);
   844→  const [busy, setBusy] = useState(false);
   845→
   846→  const attachedIds = useMemo(
   847→    () => new Set(story.media.map((m) => m.mediaId)),
   848→    [story.media]
   849→  );
   850→
   851→  const candidates = useMemo(() => {
   852→    const q = query.trim().toLowerCase();
   853→    return media
   854→      .filter((m) => !attachedIds.has(m.id))
   855→      .filter((m) => {
   856→        if (!q) return true;
   857→        return `${m.title} ${m.description} ${m.photographer} ${m.source}`.toLowerCase().includes(q);
   858→      });
   859→  }, [media, query, attachedIds]);
   860→
   861→  const doAttach = async () => {
   862→    if (!selectedId) return;
   863→    setBusy(true);
   864→    try {
   865→      await attachMedia(story.id, selectedId, {
   866→        caption: caption.trim() || null,
   867→        isPrimary: story.media.length === 0,
   868→      });
   869→      toast({ title: "Media attached to story" });
   870→      onOpenChange(false);
   871→      setCaption("");
   872→      setSelectedId(null);
   873→      setQuery("");
   874→    } catch (e) {
   875→      toast({
   876→        title: "Could not attach media",
   877→        description: e instanceof Error ? e.message : "Unknown error",
   878→        variant: "destructive",
   879→      });
   880→    } finally {
   881→      setBusy(false);
   882→    }
   883→  };
   884→
   885→  return (
   886→    <Dialog open={open} onOpenChange={onOpenChange}>
   887→      <DialogContent className="max-w-2xl">
   888→        <DialogHeader>
   889→          <DialogTitle className="font-headline">Attach media to story</DialogTitle>
   890→          <DialogDescription>
   891→            Pick an asset from the vault. Rights are re-checked automatically before publication.
   892→          </DialogDescription>
   893→        </DialogHeader>
   894→
   895→        <div className="relative">
   896→          <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   897→          <Input
   898→            value={query}
   899→            onChange={(e) => setQuery(e.target.value)}
   900→            placeholder="Search the vault…"
   901→            className="pl-8"
   902→          />
   903→        </div>
   904→
   905→        <div className="scrollbar-slim grid max-h-72 grid-cols-2 gap-2 overflow-y-auto sm:grid-cols-3">
   906→          {candidates.map((m) => (
   907→            <button
   908→              key={m.id}
   909→              onClick={() => setSelectedId(m.id)}
   910→              className={cn(
   911→                "rounded-lg border p-1.5 text-left transition-all",
   912→                selectedId === m.id
   913→                  ? "border-primary ring-1 ring-primary/50"
   914→                  : "hover:border-foreground/30"
   915→              )}
   916→            >
   917→              <div className="relative aspect-[3/2] overflow-hidden rounded bg-muted">
   918→                <Image src={m.url} alt={m.title} fill sizes="200px" className="object-cover" />
   919→              </div>
   920→              <p className="mt-1 line-clamp-1 text-xs font-medium">{m.title}</p>
   921→              <div className="mt-1 flex items-center justify-between gap-1">
   922→                <RightsBadge status={m.rights.status} label={m.rights.status === "PERPETUAL" ? "Perpetual" : undefined} />
   923→              </div>
   924→            </button>
   925→          ))}
   926→          {candidates.length === 0 && (
   927→            <p className="col-span-full py-8 text-center text-sm text-muted-foreground">
   928→              Every vault asset is already attached, or nothing matches.
   929→            </p>
   930→          )}
   931→        </div>
   932→
   933→        <div className="space-y-1.5">
   934→          <Label htmlFor="caption">Caption</Label>
   935→          <Input
   936→            id="caption"
   937→            value={caption}
   938→            onChange={(e) => setCaption(e.target.value)}
   939→            placeholder="Editorial caption shown with the image…"
   940→          />
   941→        </div>
   942→
   943→        <DialogFooter>
   944→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   945→            Cancel
   946→          </Button>
   947→          <Button onClick={doAttach} disabled={!selectedId || busy}>
   948→            {busy ? "Attaching…" : "Attach to Story"}
   949→          </Button>
   950→        </DialogFooter>
   951→      </DialogContent>
   952→    </Dialog>
   953→  );
   954→}
