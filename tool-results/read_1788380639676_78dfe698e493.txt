     3→/**
     4→ * WebMCP tool definitions for the Meridian Ledger newsroom.
     5→ *
     6→ * These tools are the machine-facing surface of the exact same REST API the
     7→ * human UI uses, so an external browser agent mutating a story through
     8→ * `update_story_status` changes the same state a human editor sees — and
     9→ * vice versa. After every mutating call we emit a `newsroom:data-changed`
    10→ * window event so the on-page UI refreshes immediately.
    11→ */

    20→function notifyDataChanged() {
    21→  window.dispatchEvent(new CustomEvent(DATA_CHANGED_EVENT));
    22→}

    24→/** Compact story projection used in list responses to keep outputs small. */
    25→function storySummary(s: Awaited<ReturnType<typeof api.fetchStories>>["stories"][number]) {
    26→  return {
    27→    id: s.id,
    28→    slug: s.slug,
    29→    title: s.title,
    30→    status: s.status,
    31→    desk: s.desk,
    32→    author: s.author,
    33→    assignee: s.assignee,
    34→    priority: s.priority,
    35→    wordCount: s.wordCount,
    36→    runOrder: s.runOrder,
    37→    mediaCount: s.media.length,
    38→    hasExpiredMedia: s.media.some((m) => m.media.rights.status === "EXPIRED"),
    39→    handoffCount: s.handoffCount,
    40→    updatedAt: s.updatedAt,
    41→  };
    42→}

    62→const ok = (data: Record<string, unknown>) => ({ ok: true as const, ...data });
    63→const fail = (error: string, extra: Record<string, unknown> = {}) => ({
    64→  ok: false as const,
    65→  error,
    66→  ...extra,
    67→});
    68→
    69→function asString(v: unknown): string | undefined {
    70→  return typeof v === "string" && v.length > 0 ? v : undefined;
    71→}
    72→
    73→function requireString(input: Record<string, unknown>, key: string): string {
    74→  const v = input[key];
    75→  if (typeof v !== "string" || v.trim() === "") {
    76→    throw new Error(`Missing required string parameter "${key}".`);
    77→  }
    78→  return v.trim();
    79→}

    81→function run<T>(fn: () => Promise<T>): Promise<T> {
    82→  return fn();
    83→}
