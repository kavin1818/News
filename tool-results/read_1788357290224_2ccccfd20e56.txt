    74→export function StoryDesk({ onSendToHandoff }: { onSendToHandoff: (id: string) => void }) {
    75→  const stories = useNewsroom((s) => s.stories);
    76→  const loading = useNewsroom((s) => s.storiesLoading);
    77→  // Desk layout + selection persist in the URL (?layout=&story=…) so views are
    78→  // deep-linkable, survive reloads, and no longer reset when switching sections.
    79→  const [layoutParam, setLayoutParam] = useQueryParam("layout", "list");
    80→  const layout: "list" | "board" = layoutParam === "board" ? "board" : "list";
    81→  const [storyParam, setStoryParam] = useQueryParam("story", "");
    82→  const [query, setQuery] = useState("");
    83→  const [statusFilter, setStatusFilter] = useState("any");
    84→  const [deskFilter, setDeskFilter] = useState("any");
    85→  const [newOpen, setNewOpen] = useState(false);
    86→  const selectedId = storyParam || null;
    87→
    88→  const filtered = useMemo(() => {
    89→    const q = query.trim().toLowerCase();
    90→    return stories.filter((s) => {
    91→      if (statusFilter !== "any" && s.status !== statusFilter) return false;
    92→      if (deskFilter !== "any" && s.desk !== deskFilter) return false;
    93→      if (q) {
    94→        const hay = `${s.title} ${s.summary} ${s.author} ${s.assignee ?? ""} ${s.slug}`.toLowerCase();
    95→        if (!hay.includes(q)) return false;
    96→      }
    97→      return true;
    98→    });
    99→  }, [stories, query, statusFilter, deskFilter]);
   100→
   101→  const selected =
   102→    (selectedId ? stories.find((s) => s.id === selectedId) : undefined) ??
   103→    filtered[0] ??
   104→    null;
   105→
   106→  const statusCounts = useMemo(() => {
   107→    const counts: Record<string, number> = { any: stories.length };
   108→    for (const s of stories) counts[s.status] = (counts[s.status] ?? 0) + 1;
   109→    return counts;
   110→  }, [stories]);
   111→
   112→  return (
   113→    <div className="rise-in">
   114→      <div className="mb-5 flex flex-wrap items-end justify-between gap-3">
   115→        <div>
   116→          <h2 className="font-headline text-2xl font-bold">Story Desk</h2>
   117→          <p className="text-sm text-muted-foreground">
   118→            Write, assign, review and move stories through the editorial workflow.
   119→          </p>
   120→        </div>
   121→        <div className="flex items-center gap-2">
   122→          <div
   123→            role="group"
   124→            aria-label="Desk layout"
   125→            className="flex rounded-md border bg-card p-0.5 shadow-sm"
   126→          >
   127→            <button
   128→              type="button"
   129→              onClick={() => setLayoutParam("list")}
   130→              aria-pressed={layout === "list"}
   131→              title="List view"
   132→              className={cn(
   133→                "flex items-center gap-1.5 rounded-sm px-2.5 py-1.5 text-xs font-medium transition-colors",
   134→                layout === "list"
   135→                  ? "bg-primary text-primary-foreground"
   136→                  : "text-muted-foreground hover:text-foreground"
   137→              )}
   138→            >
   139→              <LayoutList className="h-3.5 w-3.5" />
   140→              <span className="hidden sm:inline">List</span>
   141→            </button>
   142→            <button
   143→              type="button"
   144→              onClick={() => setLayoutParam("board")}
   145→              aria-pressed={layout === "board"}
   146→              title="Board view — stories grouped by workflow stage"
   147→              className={cn(
   148→                "flex items-center gap-1.5 rounded-sm px-2.5 py-1.5 text-xs font-medium transition-colors",
   149→                layout === "board"
   150→                  ? "bg-primary text-primary-foreground"
   151→                  : "text-muted-foreground hover:text-foreground"
   152→              )}
   153→            >
   154→              <KanbanSquare className="h-3.5 w-3.5" />
   155→              <span className="hidden sm:inline">Board</span>
   156→            </button>
   157→          </div>
   158→          <Button onClick={() => setNewOpen(true)} className="gap-1.5">
   159→            <Plus className="h-4 w-4" /> New Story
   160→          </Button>
   161→        </div>
   162→      </div>
   163→
   164→      {layout === "board" ? (
   165→        <StoryBoard
   166→          stories={filtered}
   167→          loading={loading && stories.length === 0}
   168→          onOpen={(id) => {
   169→            setStoryParam(id);
   170→            setLayoutParam("list");
   171→          }}
   172→          onSendToHandoff={onSendToHandoff}
   173→        />
   174→      ) : (
   175→      <div className="grid gap-6 lg:grid-cols-5">
   176→        {/* ------------------------------------------------------ list pane */}
   177→        <section aria-label="Story list" className="min-w-0 lg:col-span-2">
   178→          <div className="flex gap-2">
   179→            <div className="relative flex-1">
   180→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   181→              <Input
   182→                value={query}
   183→                onChange={(e) => setQuery(e.target.value)}
   184→                placeholder="Search stories…"
   185→                className="pl-8"
   186→                aria-label="Search stories"
   187→              />
   188→            </div>
   189→            <Select value={deskFilter} onValueChange={setDeskFilter}>
   190→              <SelectTrigger className="w-[140px]" aria-label="Filter by desk">
   191→                <SelectValue placeholder="All desks" />
   192→              </SelectTrigger>
   193→              <SelectContent>
   194→                <SelectItem value="any">All desks</SelectItem>
   195→                {DESKS.map((d) => (
   196→                  <SelectItem key={d} value={d}>
   197→                    {d}
   198→                  </SelectItem>
   199→                ))}
   200→              </SelectContent>
   201→            </Select>
   202→          </div>
   203→
   204→          <div className="scrollbar-slim mt-3 flex flex-wrap gap-1.5 pb-1">
   205→            {[
   206→              { id: "any", label: "All" },
   207→              { id: "DRAFT", label: "Draft" },
   208→              { id: "IN_REVIEW", label: "In Review" },
   209→              { id: "REVISION_REQUESTED", label: "Revision" },
   210→              { id: "APPROVED", label: "Approved" },
   211→              { id: "PUBLISHED", label: "Published" },
   212→            ].map((st) => (
   213→              <button
   214→                key={st.id}
   215→                onClick={() => setStatusFilter(st.id)}
   216→                className={cn(
   217→                  "whitespace-nowrap rounded-full border px-2.5 py-1 text-xs font-medium transition-all active:scale-95",
   218→                  statusFilter === st.id
   219→                    ? "border-primary bg-primary text-primary-foreground shadow-sm"
   220→                    : "bg-card text-muted-foreground hover:border-foreground/30 hover:text-foreground"
   221→                )}
   222→              >
   223→                {st.label}
   224→                <span className={cn("ml-1", statusFilter === st.id ? "text-primary-foreground/80" : "text-muted-foreground/70")}>
   225→                  {statusCounts[st.id] ?? 0}
   226→                </span>
   227→              </button>
   228→            ))}
   229→          </div>
   230→
   231→          <div className="scrollbar-slim mt-3 max-h-[calc(100vh-16rem)] space-y-2 overflow-y-auto pr-1 lg:max-h-[calc(100vh-19rem)]">
   232→            {loading && stories.length === 0 && (
   233→              <div className="space-y-2">
   234→                {[0, 1, 2, 3].map((i) => (
   235→                  <div key={i} className="flex items-start gap-3 rounded-lg border p-3">
   236→                    <Skeleton className="h-14 w-20 rounded-md" />
   237→                    <div className="flex-1 space-y-2">
   238→                      <Skeleton className="h-3.5 w-4/5" />
   239→                      <Skeleton className="h-3 w-1/3" />
   240→                      <Skeleton className="h-2.5 w-1/2" />
   241→                    </div>
   242→                  </div>
   243→                ))}
   244→              </div>
   245→            )}
   246→            {!loading && filtered.length === 0 && (
   247→              <div className="flex flex-col items-center gap-3 rounded-lg border border-dashed bg-card/50 py-10 text-center">
   248→                <FileText className="h-7 w-7 text-muted-foreground/40" />
   249→                <p className="text-sm text-muted-foreground">
   250→                  No stories match the current search and filters.
   251→                </p>
   252→                {(query || statusFilter !== "any" || deskFilter !== "any") && (
   253→                  <Button
   254→                    size="sm"
   255→                    variant="outline"
   256→                    onClick={() => {
   257→                      setQuery("");
   258→                      setStatusFilter("any");
   259→                      setDeskFilter("any");
   260→                    }}
   261→                  >
   262→                    <X className="mr-1.5 h-3.5 w-3.5" /> Clear filters
   263→                  </Button>
   264→                )}
   265→              </div>
   266→            )}
   267→            {filtered.map((story) => (
   268→              <StoryListItem
   269→                key={story.id}
   270→                story={story}
   271→                active={story.id === selectedId}
   272→                onSelect={() => setStoryParam(story.id)}
   273→              />
   274→            ))}
   275→          </div>
   276→        </section>
   277→
   278→        {/* ----------------------------------------------------- detail pane */}
   279→        <section aria-label="Story detail" className="min-w-0 lg:col-span-3">
   280→          {selected ? (
   281→            <StoryDetail
   282→              key={selected.id}
   283→              story={selected}
   284→              onSendToHandoff={onSendToHandoff}
   285→              onDuplicated={setStoryParam}
   286→            />
   287→          ) : (
   288→            <div className="flex h-64 flex-col items-center justify-center gap-2 rounded-lg border border-dashed bg-card/50 text-center">
   289→              <FileText className="h-8 w-8 text-muted-foreground/50" />
   290→              <p className="text-sm text-muted-foreground">
   291→                Select a story from the desk, or create a new one.
   292→              </p>
   293→            </div>
   294→          )}
   295→        </section>
   296→      </div>
   297→      )}
   298→
   299→      <NewStoryDialog open={newOpen} onOpenChange={setNewOpen} onCreated={(id) => setStoryParam(id)} />
   300→    </div>
   301→  );
   302→}

   309→  story,
   310→  active,
   311→  onSelect,
   312→}: {
   313→  story: StoryDTO;
   314→  active: boolean;
   315→  onSelect: () => void;
   316→}) {
   317→  const primary = story.media.find((m) => m.isPrimary) ?? story.media[0];
   318→  const hasExpired = story.media.some((m) => m.media.rights.status === "EXPIRED");
   319→
   320→  return (
   321→    <button
   322→      onClick={onSelect}
   323→      aria-current={active ? "true" : undefined}
   324→      className={cn(
   325→        "group flex w-full items-start gap-3 rounded-lg border p-3 text-left transition-all active:scale-[0.99]",
   326→        active
   327→          ? "border-primary/60 bg-card shadow-[0_1px_0_0_var(--primary)] ring-1 ring-primary/40"
   328→          : "border-border bg-card hover:border-foreground/25 hover:shadow-sm"
   329→      )}
   330→    >
   331→      <div className="relative h-14 w-20 shrink-0 overflow-hidden rounded-md bg-muted">
   332→        {primary ? (
   333→          <Image
   334→            src={primary.media.url}
   335→            alt={primary.media.title}
   336→            fill
   337→            sizes="80px"
   338→            className="object-cover"
   339→          />
   340→        ) : (
   341→          <div className="flex h-full w-full items-center justify-center">
   342→            <ImageIcon className="h-4 w-4 text-muted-foreground/40" />
   343→          </div>
   344→        )}
   345→      </div>
   346→      <div className="min-w-0 flex-1">
   347→        <p className="line-clamp-2 font-headline text-[15px] font-semibold leading-snug transition-colors group-hover:text-primary">
   348→          {story.title}
   349→        </p>
   350→        <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   351→          <StatusBadge status={story.status} />
   352→          {(story.priority === "HIGH" || story.priority === "URGENT") && (
   353→            <Badge
   354→              variant="outline"
   355→              className={cn(
   356→                "px-1.5 py-0 text-[10px] font-bold",
   357→                story.priority === "URGENT"
   358→                  ? "border-red-300 bg-red-50 text-red-700 dark:border-red-800 dark:bg-red-950/50 dark:text-red-300"
   359→                  : "border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-800 dark:bg-amber-950/50 dark:text-amber-300"
   360→              )}
   361→            >
   362→              {story.priority}
   363→            </Badge>
   364→          )}
   365→        </div>
   366→        <p className="mt-1.5 flex flex-wrap items-center gap-x-2 text-xs text-muted-foreground">
   367→          <span className="flex items-center gap-1.5 font-medium text-foreground/70">
   368→            <DeskDot desk={story.desk} />
   369→            {story.desk}
   370→          </span>
   371→          <span aria-hidden>·</span>
   372→          <span>{story.author}</span>
   373→          <span aria-hidden>·</span>
   374→          <span>{fmtRelative(story.updatedAt)}</span>
   375→          {hasExpired && (
   376→            <span className="inline-flex items-center gap-0.5 font-medium text-red-700 dark:text-red-400">
   377→              <AlertTriangle className="h-3 w-3" /> expired media
   378→            </span>
   379→          )}
   380→          {story.scheduledFor &&
   381→            story.status !== "PUBLISHED" &&
   382→            deadlineState(story.scheduledFor) === "past" && (
   383→              <span className="inline-flex items-center gap-0.5 font-medium text-red-700 dark:text-red-400">
   384→                <CalendarClock className="h-3 w-3" /> past due
   385→              </span>
   386→            )}
   387→          {story.scheduledFor &&
   388→            story.status !== "PUBLISHED" &&
   389→            deadlineState(story.scheduledFor) === "soon" && (
   390→              <span className="inline-flex items-center gap-0.5 font-medium text-amber-700 dark:text-amber-400">
   391→                <CalendarClock className="h-3 w-3" /> due soon
   392→              </span>
   393→            )}
   394→        </p>
   395→      </div>
   396→    </button>
   397→  );
   398→}

   400→// ---------------------------------------------------------------------------
   401→// Desk board — editorial run-order board, stories grouped by workflow stage.
   402→// ---------------------------------------------------------------------------
   403→
   404→const BOARD_QUICK_ACTION: Partial<
   405→  Record<(typeof STORY_STATUSES)[number], { label: string; icon: React.ComponentType<{ className?: string }> }>
   406→> = {
   407→  DRAFT: { label: "Submit", icon: Eye },
   408→  IN_REVIEW: { label: "Approve", icon: CheckCheck },
   409→  REVISION_REQUESTED: { label: "Resubmit", icon: Eye },
   410→  APPROVED: { label: "Handoff", icon: Send },
   411→};
   412→
   413→function StoryBoard({
   414→  stories,
   415→  loading,
   416→  onOpen,
   417→  onSendToHandoff,
   418→}: {
   419→  stories: StoryDTO[];
   420→  loading: boolean;
   421→  onOpen: (id: string) => void;
   422→  onSendToHandoff: (id: string) => void;
   423→}) {
   424→  const setStoryStatus = useNewsroom((s) => s.setStoryStatus);
   425→  const approveStory = useNewsroom((s) => s.approveStory);
   426→  const { toast } = useToast();
   427→  const [busyId, setBusyId] = useState<string | null>(null);
   428→  const [dragId, setDragId] = useState<string | null>(null);
   429→  const [overStatus, setOverStatus] = useState<string | null>(null);
   430→
   431→  const dragStory = dragId ? (stories.find((s) => s.id === dragId) ?? null) : null;
   432→  const legalTargets = dragStory ? (STATUS_TRANSITIONS[dragStory.status as (typeof STORY_STATUSES)[number]] ?? []) : [];
   433→
   434→  const attemptMove = async (story: StoryDTO, target: string) => {
   435→    if (story.status === target) return;
   436→    const legal = (STATUS_TRANSITIONS[story.status as (typeof STORY_STATUSES)[number]] ?? []).includes(
   437→      target as (typeof STORY_STATUSES)[number]
   438→    );
   439→    if (!legal) {
   440→      toast({
   441→        title: "Move not allowed",
   442→        description: `${STATUS_LABELS_UI[story.status as (typeof STORY_STATUSES)[number]]} → ${STATUS_LABELS_UI[target as (typeof STORY_STATUSES)[number]]} is outside the workflow. Publishing goes through the Handoff Engine.`,
   443→        variant: "destructive",
   444→      });
   445→      return;
   446→    }
   447→    setBusyId(story.id);
   448→    try {
   449→      if (target === "APPROVED") {
   450→        await approveStory(story.id);
   451→        toast({ title: "Story approved" });
   452→      } else {
   453→        await setStoryStatus(story.id, target);
   454→        toast({ title: `Moved to ${STATUS_LABELS_UI[target as (typeof STORY_STATUSES)[number]]}` });
   455→      }
   456→    } catch (e) {
   457→      toast({
   458→        title: "Move failed",
   459→        description: e instanceof Error ? e.message : "Unknown error",
   460→        variant: "destructive",
   461→      });
   462→    } finally {
   463→      setBusyId(null);
   464→    }
   465→  };
   466→
   467→  const runQuick = async (story: StoryDTO) => {
   468→    setBusyId(story.id);
   469→    try {
   470→      if (story.status === "DRAFT") {
   471→        await setStoryStatus(story.id, "IN_REVIEW");
   472→        toast({ title: "Story submitted for review" });
   473→      } else if (story.status === "IN_REVIEW") {
   474→        await approveStory(story.id);
   475→        toast({ title: "Story approved" });
   476→      } else if (story.status === "REVISION_REQUESTED") {
   477→        await setStoryStatus(story.id, "IN_REVIEW");
   478→        toast({ title: "Resubmitted for review" });
   479→      } else if (story.status === "APPROVED") {
   480→        onSendToHandoff(story.id);
   481→        return;
   482→      }
   483→    } catch (e) {
   484→      toast({
   485→        title: "Action failed",
   486→        description: e instanceof Error ? e.message : "Unknown error",
   487→        variant: "destructive",
   488→      });
   489→    } finally {
   490→      setBusyId(null);
   491→    }
   492→  };
   493→
   494→  return (
   495→    <div className="rise-in">
   496→      <p className="mb-3 text-xs text-muted-foreground">
   497→        Run-order board — the desk at a glance. Drag cards between stages
   498→        (workflow rules apply), click to open, or use the inline next step.
   499→      </p>
   500→      {loading ? (
   501→        <div className="flex gap-3">
   502→          {STORY_STATUSES.map((s) => (
   503→            <div key={s} className="board-column w-[230px] shrink-0 space-y-2 pb-2">
   504→              <Skeleton className="h-6 w-full" />
   505→              <Skeleton className="h-24 w-full" />
   506→              <Skeleton className="h-24 w-full" />
   507→            </div>
   508→          ))}
   509→        </div>
   510→      ) : (
   511→        <div className="scrollbar-slim flex gap-3 overflow-x-auto pb-2">
   512→          {STORY_STATUSES.map((status) => {
   513→            const column = stories.filter((s) => s.status === status);
   514→            const isSource = dragStory?.status === status;
   515→            const isLegalTarget = legalTargets.includes(status);
   516→            const isOver = overStatus === status;
   517→            return (
   518→              <section
   519→                key={status}
   520→                aria-label={`${STATUS_LABELS_UI[status]} column`}
   521→                onDragOver={(e) => {
   522→                  // Gate on the drag payload, not React state: dragstart may
   523→                  // not have flushed state yet when the first dragover fires.
   524→                  if (!e.dataTransfer.types.includes("text/plain")) return;
   525→                  e.preventDefault();
   526→                  e.dataTransfer.dropEffect = "move";
   527→                  if (overStatus !== status) setOverStatus(status);
   528→                }}
   529→                onDragLeave={(e) => {
   530→                  if (!e.currentTarget.contains(e.relatedTarget as Node)) {
   531→                    setOverStatus((s) => (s === status ? null : s));
   532→                  }
   533→                }}
   534→                onDrop={(e) => {
   535→                  e.preventDefault();
   536→                  const id = e.dataTransfer.getData("text/plain") || dragId;
   537→                  const story = id ? stories.find((s) => s.id === id) : null;
   538→                  setOverStatus(null);
   539→                  setDragId(null);
   540→                  if (story) void attemptMove(story, status);
   541→                }}
   542→                className={cn(
   543→                  "board-column w-[236px] shrink-0 rounded-b-lg transition-all sm:w-[252px]",
   544→                  dragStory && isLegalTarget &&

  1052→}: {
  1053→  story: StoryDTO;

  1066→      await detachMedia(story.id, mediaId);

  1071→        description: e instanceof Error ? e.message : "Unknown error",

  1075→      setBusy(null);
  1076→    }

  1079→  const makePrimary = async (mediaId: string) => {
  1080→    setBusy(mediaId);
  1081→    try {
  1082→      await updateAttachment(story.id, mediaId, { isPrimary: true });
  1083→      toast({ title: "Lead image updated" });
  1084→    } catch (e) {
  1085→      toast({
  1086→        title: "Could not set lead image",
  1087→        description: e instanceof Error ? e.message : "Unknown error",
  1088→        variant: "destructive",
  1089→      });
  1090→    } finally {
  1091→      setBusy(null);
  1092→    }
  1093→  };
  1094→

  1218→function AttachMediaDialog({
  1219→  story,
  1220→  open,
  1221→  onOpenChange,
  1222→}: {
  1223→  story: StoryDTO;
  1224→  open: boolean;
  1225→  onOpenChange: (v: boolean) => void;
  1226→}) {
  1227→  const media = useNewsroom((s) => s.media);
  1228→  const attachMedia = useNewsroom((s) => s.attachMedia);
  1229→  const { toast } = useToast();
  1230→  const [query, setQuery] = useState("");
  1231→  const [caption, setCaption] = useState("");
  1232→  const [selectedId, setSelectedId] = useState<string | null>(null);
  1233→  const [busy, setBusy] = useState(false);
  1234→
  1235→  const attachedIds = useMemo(
  1236→    () => new Set(story.media.map((m) => m.mediaId)),
  1237→    [story.media]
  1238→  );
  1239→
  1240→  const candidates = useMemo(() => {
  1241→    const q = query.trim().toLowerCase();
  1242→    return media
  1243→      .filter((m) => !attachedIds.has(m.id))
  1244→      .filter((m) => {
  1245→        if (!q) return true;
  1246→        return `${m.title} ${m.description} ${m.photographer} ${m.source}`.toLowerCase().includes(q);
  1247→      });
  1248→  }, [media, query, attachedIds]);
  1249→
  1250→  const doAttach = async () => {
  1251→    if (!selectedId) return;
  1252→    setBusy(true);
  1253→    try {
  1254→      await attachMedia(story.id, selectedId, {
  1255→        caption: caption.trim() || null,
  1256→        isPrimary: story.media.length === 0,
  1257→      });
  1258→      toast({ title: "Media attached to story" });
  1259→      onOpenChange(false);
  1260→      setCaption("");
  1261→      setSelectedId(null);
  1262→      setQuery("");
  1263→    } catch (e) {
  1264→      toast({
  1265→        title: "Could not attach media",
  1266→        description: e instanceof Error ? e.message : "Unknown error",
  1267→        variant: "destructive",
  1268→      });
  1269→    } finally {
  1270→      setBusy(false);
  1271→    }
  1272→  };
  1273→
  1274→  return (
  1275→    <Dialog open={open} onOpenChange={onOpenChange}>
  1276→      <DialogContent className="max-w-2xl">
  1277→        <DialogHeader>
  1278→          <DialogTitle className="font-headline">Attach media to story</DialogTitle>
  1279→          <DialogDescription>
  1280→            Pick an asset from the vault. Rights are re-checked automatically before publication.
  1281→          </DialogDescription>
  1282→        </DialogHeader>
  1283→
  1284→        <div className="relative">
  1285→          <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
  1286→          <Input
  1287→            value={query}
  1288→            onChange={(e) => setQuery(e.target.value)}
  1289→            placeholder="Search the vault…"
  1290→            className="pl-8"
  1291→          />
  1292→        </div>
  1293→
  1294→        <div className="scrollbar-slim grid max-h-72 grid-cols-2 gap-2 overflow-y-auto sm:grid-cols-3">
  1295→          {candidates.map((m) => (
  1296→            <button
  1297→              key={m.id}
  1298→              onClick={() => setSelectedId(m.id)}
  1299→              className={cn(
  1300→                "rounded-lg border p-1.5 text-left transition-all",
  1301→                selectedId === m.id
  1302→                  ? "border-primary ring-1 ring-primary/50"
  1303→                  : "hover:border-foreground/30"
  1304→              )}
  1305→            >
  1306→              <div className="relative aspect-[3/2] overflow-hidden rounded bg-muted">
  1307→                <Image src={m.url} alt={m.title} fill sizes="200px" className="object-cover" />
  1308→              </div>
  1309→              <p className="mt-1 line-clamp-1 text-xs font-medium">{m.title}</p>
  1310→              <div className="mt-1 flex items-center justify-between gap-1">
  1311→                <RightsBadge status={m.rights.status} label={m.rights.status === "PERPETUAL" ? "Perpetual" : undefined} />
  1312→              </div>
  1313→            </button>
  1314→          ))}
  1315→          {candidates.length === 0 && (
  1316→            <p className="col-span-full py-8 text-center text-sm text-muted-foreground">
  1317→              Every vault asset is already attached, or nothing matches.
  1318→            </p>
  1319→          )}
  1320→        </div>
  1321→
  1322→        <div className="space-y-1.5">
  1323→          <Label htmlFor="caption">Caption</Label>
  1324→          <Input
  1325→            id="caption"
  1326→            value={caption}

  1328→            placeholder="Editorial caption shown with the image…"
  1329→          />
  1330→        </div>
  1331→
  1332→        <DialogFooter>
  1333→          <Button variant="outline" onClick={() => onOpenChange(false)}>
  1334→            Cancel
  1335→          </Button>
  1336→          <Button onClick={doAttach} disabled={!selectedId || busy}>
  1337→            {busy ? "Attaching…" : "Attach to Story"}
  1338→          </Button>
  1339→        </DialogFooter>
  1340→      </DialogContent>
  1341→    </Dialog>
  1342→  );
  1343→}
